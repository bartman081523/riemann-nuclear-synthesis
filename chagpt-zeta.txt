import math


def P_X_zeta23(X, tau):
    """
    Zeta23.Defs.lean:

      P_X(tau)
        = -(1/pi) sum_{n <= X}
          Lambda(n)/sqrt(n) cos(tau log n)
    """
    n_max = int(np.floor(X))

    total = 0.0

    for n in range(2, n_max + 1):
        lam = von_mangoldt(n)

        if lam != 0.0:
            total += (
                lam /
                math.sqrt(n) *
                math.cos(tau * math.log(n))
            )

    return -total / np.pi


def nu_X_zeta23(X, tau):
    """
    Exact numerical translation of:

        nu_X = mu + Pi_X + P_X
    """
    return (
        mu_zeta23(tau)
        + Pi_X_zeta23(X, tau)
        + P_X_zeta23(X, tau)
    )
